import math
import numpy as np

Z = np.linspace(0.1, 10, 100)
B = 0.38
R = 1.75
D = 4

def flux(z):
    return (((B/2)*( ( (D+z) / math.sqrt((R**2) + ((D+z)**2)) )  -  ( z / math.sqrt((R**2)+(z**2)) )))*1000)

def voltage(x):
    return (600 + x * (100))

def value(x):
    return x / (3300.0 / 256.0)

flux_vec = np.vectorize(flux)
voltage_vec = np.vectorize(voltage)
value_vec = np.vectorize(value)

prev = 0
for z in Z:
    z = round(z, 2)
    y = float(flux_vec(z))
    v = float(voltage_vec(y))
    value = float(value_vec(v))
    value = round(value)
    print(str(value) + "..=" + str(prev-1) + " => " + str(int(z * 10)) + ",")
    prev = value
